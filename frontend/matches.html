from sklearn.feature_extraction.text import TfidfVectorizer from sklearn.metrics.pairwise import cosine_similarity from datetime import datetime def text_similarity(a: str, b: str) -> float: a = (a or "").strip() b = (b or "").strip() if not a or not b: return 0.0 try: vectorizer = TfidfVectorizer(stop_words="english") vectors = vectorizer.fit_transform([a, b]) score = cosine_similarity(vectors[0:1], vectors[1:2])[0][0] return float(score) except ValueError: return 0.0 def location_score(loc_a: str, loc_b: str) -> float: loc_a = (loc_a or "").strip().lower() loc_b = (loc_b or "").strip().lower() if not loc_a or not loc_b: return 0.0 return 1.0 if loc_a == loc_b else 0.3 def time_score(time_a: str, time_b: str) -> float: try: dt_a = datetime.fromisoformat(time_a) dt_b = datetime.fromisoformat(time_b) minutes = abs((dt_a - dt_b).total_seconds()) / 60 except (ValueError, TypeError): return 0.0 if minutes <= 30: return 1.0 if minutes <= 120: return 0.8 if minutes <= 1440: return 0.5 return 0.2 def final_score(text_score: float, loc_score: float, t_score: float) -> float: return round((0.5 * text_score) + (0.25 * loc_score) + (0.25 * t_score), 3)
